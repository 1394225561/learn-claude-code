指数退避

def with_retry(fn, state):
    for attempt in range(5):
        try:
            return fn()
        except RateLimitError as e:
            delay = min(500 * (2 ** attempt), 32000)  # 500ms → 1s → 2s → 4s → 8s
            delay += random.uniform(0, delay * 0.25)   # 加抖动，避免雷群效应
            time.sleep(delay / 1000)

            if is_529(e):
                state.consecutive_529 += 1
                if state.consecutive_529 >= 3:
                    state.current_model = FALLBACK_MODEL  # 切换备用模型
    raise  # 所有重试都失败后，抛出最后一个异常

RecoveryState

class RecoveryState:
    def __init__(self):
        self.consecutive_529 = 0
        self.current_model = PRIMARY_MODEL
        self.max_tokens_recoveries = 0
        self.has_compacted = False
