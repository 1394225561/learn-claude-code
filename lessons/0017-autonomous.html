空闲轮询

IDLE_TIMEOUT = 60       # 空闲 60 秒后退出
IDLE_POLL_INTERVAL = 5  # 每 5 秒检查一次

def idle_poll(agent_name, messages, name, role) -> str:
    """WORK → IDLE：主动寻找新任务"""
    for _ in range(IDLE_TIMEOUT // IDLE_POLL_INTERVAL):
        time.sleep(IDLE_POLL_INTERVAL)

        # 优先级 1：检查收件箱（Lead 可能有新指令）
        inbox = BUS.read_inbox(agent_name)
        if inbox:
            return "work"  # 有消息，回到工作状态

        # 优先级 2：扫描任务板
        unclaimed = scan_unclaimed_tasks()
        if unclaimed:
            claim_task(unclaimed[0]["id"], agent_name)
            return "work"  # 找到任务，开始工作

    return "timeout"  # 空闲太久，准备退出

扫描未认领任务

def scan_unclaimed_tasks():
    """找 pending、无 owner、依赖已满足的任务"""
    tasks = load_all_tasks()
    return [
        t for t in tasks
        if t["status"] == "pending"
        and not t.get("owner")        # 没人认领
        and can_start(t["id"])        # 依赖都完成了
    ]

前端类比

这就是 Worker Pool + 任务队列模式。空闲轮询像 Service Worker 的 periodicsync，自动认领像 Redis 分布式锁或 Bull/BullMQ 任务队列。
